namespace = rush

event = {
	id = rush.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		NOT = { has_global_flag = rush_enabled }
	}
	immediate = {
		every_country = {
			limit = 
				{ NOT = { has_modifier = rush_event_modifier } }
				country_event = { id = rush.2 }
		}
		set_global_flag = rush_enabled
	}

}

country_event = {
	id = rush.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_modifier = rush_event_modifier } }
			add_modifier = {
				modifier = rush_event_modifier
				days = -1
			}
		}
	}
}
